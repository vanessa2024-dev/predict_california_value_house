// --- Fonction de validation mfValidation() ---
function mfValidation() {
    const username = document.getElementById('username').value.trim();
    const password = document.getElementById('password').value.trim();

    if (username === "" || password === "") {
        alert("Veuillez remplir tous les champs !");
        return false; // Empêche l'envoi du formulaire
    }

    return true; // Formulaire OK
}

// --- Ajouter un checkbox pour afficher/masquer le mot de passe ---
const passwordInput = document.getElementById('password');

// Créer le checkbox et le label
const showPwdCheckbox = document.createElement('input');
showPwdCheckbox.type = 'checkbox';
showPwdCheckbox.id = 'showPwd';

const showPwdLabel = document.createElement('label');
showPwdLabel.htmlFor = 'showPwd';
showPwdLabel.textContent = 'Afficher le mot de passe';

// Ajouter le checkbox et label juste après le champ mot de passe
passwordInput.parentNode.insertBefore(showPwdCheckbox, passwordInput.nextSibling);
passwordInput.parentNode.insertBefore(showPwdLabel, showPwdCheckbox.nextSibling);

// Quand on coche/décoche le checkbox, le type de l’input change
showPwdCheckbox.addEventListener('change', function() {
    passwordInput.type = this.checked ? 'text' : 'password';
});
